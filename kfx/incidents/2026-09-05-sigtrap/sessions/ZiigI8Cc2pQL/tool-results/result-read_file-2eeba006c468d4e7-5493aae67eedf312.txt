95 	function tree(root: string, relative = ""): string[] {
96 	  const path = relative ? join(root, relative) : root;
97 	  const entries = readdirSync(path, { withFileTypes: true });
98 	  const result: string[] = [];
99 	  for (const entry of entries) {
100	    const child = relative ? join(relative, entry.name) : entry.name;
101	    result.push(child);
102	    if (entry.isDirectory()) result.push(...tree(root, child));
103	  }
104	  return result.sort();
105	}
