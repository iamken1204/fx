36 	function loadDotEnv(): Record<string, string> {
37 	  const candidates = [join(REPO_ROOT, ".env"), join(REPO_ROOT, ".env.local")];
38 	  const vars: Record<string, string> = {};
39 	  for (const file of candidates) {
40 	    if (!existsSync(file)) continue;
41 	    const lines = readFileSync(file, "utf-8").split("\n");
42 	    for (const line of lines) {
43 	      const trimmed = line.trim();
44 	      if (!trimmed || trimmed.startsWith("#")) continue;
45 	      const eq = trimmed.indexOf("=");
46 	      if (eq < 0) continue;
47 	      const key = trimmed.slice(0, eq).trim();
48 	      let value = trimmed.slice(eq + 1).trim();
49 	      if (
50 	        (value.startsWith('"') && value.endsWith('"')) ||
51 	        (value.startsWith("'") && value.endsWith("'"))
52 	      ) {
53 	        value = value.slice(1, -1);
54 	      }
55 	      vars[key] = value;
56 	    }
57 	  }
58 	  return vars;
59 	}

171	export async function runEval(
172	  prompt: string,
173	  opts: EvalOptions = {},
174	): Promise<EvalResult> {
175	  const { timeoutSec = 300, model = EVAL_MODEL, setup } = opts;
176	
177	  const workDir = opts.cwd ?? createWorkDir();
178	  const home = createEvalHome();
179	
180	  try {
181	    if (setup) {
182	      await setup(workDir);
183	    }
184	
185	    if (!existsSync(FX_BIN)) {
186	      throw new Error(
187	        `fx binary not found at ${FX_BIN}. Run 'zig build' first.`,
188	      );
189	    }
190	
191	    const args = [
192	      "ask",
193	      "--auto",
194	      "--json",
195	      "--no-save",
196	      "--timeout",
197	      String(timeoutSec * 1000),
198	      prompt,
199	    ];
200	
201	    const result = await new Promise<{
202	      stdout: string;
203	      stderr: string;
204	      code: number | null;
205	    }>((resolvePromise) => {
206	      const env = buildEvalProcessEnv(home, model);
207	      const child = nodeSpawn(FX_BIN, args, {
208	        env,
209	        cwd: workDir,
210	        stdio: ["pipe", "pipe", "pipe"],
211	      });
212	
213	      const stdoutBufs: Buffer[] = [];
214	      const stderrBufs: Buffer[] = [];
215	      child.stdout.on("data", (d: Buffer) => stdoutBufs.push(d));
216	      child.stderr.on("data", (d: Buffer) => stderrBufs.push(d));
217	      child.stdin.end();
218	
219	      child.on("close", (code: number | null) => {
220	        resolvePromise({
221	          stdout: Buffer.concat(stdoutBufs).toString(),
222	          stderr: Buffer.concat(stderrBufs).toString(),
223	          code,
224	        });
225	      });
226	    });
227	
228	    let json: HeadlessResult;
229	    try {
230	      json = JSON.parse(result.stdout.trim());
231	    } catch {
232	      throw new Error(
233	        `Failed to parse fx JSON output.\nstdout: ${result.stdout}\nstderr: ${result.stderr}`,
234	      );
235	    }
236	
237	    console.log("\n--- eval result ---");
238	    console.log(`  exit code: ${result.code}`);
239	    console.log(`  json.exit_code: ${json.exit_code}`);
240	    console.log(`  json.model: ${json.model}`);
241	    console.log(`  json.steps: ${json.steps}`);
242	    console.log(`  json.tool_calls: ${json.tool_calls?.length ?? 0}`);
243	    console.log(`  json.output length: ${json.output.length}`);
244	    if (json.error) {
245	      console.log(`  json.error: ${json.error}`);
246	      console.log(`  json.output: ${json.output.slice(0, 500)}`);
247	    }
248	    if (result.stderr) {
249	      console.log(`  stderr (last 500): ${result.stderr.slice(-500)}`);
250	    }
251	    console.log("--- end eval result ---\n");
252	
253	    if (json.error) {
254	      throw new Error(
255	        `fx returned error: ${json.error}\nstderr: ${result.stderr.slice(-1000)}`,
256	      );
257	    }
258	
259	    return {
260	      json,
261	      stdout: result.stdout,
262	      stderr: result.stderr,
263	      code: result.code,
264	      workDir,
265	    };
266	  } finally {
267	    cleanupEvalHome(home);
268	  }
269	}
